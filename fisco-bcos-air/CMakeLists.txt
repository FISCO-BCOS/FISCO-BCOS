aux_source_directory(. SRC_LIST)

set(BINARY_NAME fisco-bcos)
add_executable(${BINARY_NAME} ${SRC_LIST})
# blst: use IMPORTED_LOCATION from find_package target to avoid bare -lblst
if(TARGET blst)
    get_target_property(_blst_loc blst IMPORTED_LOCATION)
    if(_blst_loc)
        target_link_libraries(${BINARY_NAME} PUBLIC init pbft_init command gateway ${_blst_loc})
    else()
        target_link_libraries(${BINARY_NAME} PUBLIC init pbft_init command gateway blst)
    endif()
else()
    message(FATAL_ERROR "blst target not found — ensure find_package(blst) is called")
endif()
set_target_properties(${BINARY_NAME} PROPERTIES UNITY_BUILD "ON")
